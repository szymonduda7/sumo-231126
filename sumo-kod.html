1. ManagerScript

ServerScriptService → Manager

Spawnuje 3 boty w losowych miejscach na arenie, usuwa te, które spadły lub zginęły.

-- MIEJSCE: ServerScriptService > Manager  (typ: Script)
-- Spawnuje boty na arenie, usuwa te, które spadły lub zginęły.

local ServerStorage = game:GetService("ServerStorage")
local enemyTemplate = ServerStorage:WaitForChild("Enemy")
local arena = workspace:WaitForChild("Arena")

-- KONFIGURACJA
local MAX_ENEMIES = 3
local FALL_LIMIT = 15      -- ile studów poniżej areny bot jest uznany za przegranego
local SPAWN_RADIUS = 0.7   -- 0.7 = boty pojawiają się w 70% promienia (nie przy krawędzi)

-- Arena to walec postawiony pionowo: Size.X = grubość, Size.Y/Size.Z = średnica
local function getArenaTop()
	return arena.Position.Y + arena.Size.X / 2
end

-- Losowy punkt w kole (a nie w prostokącie)
local function getRandomPosition()
	local radius = math.min(arena.Size.Y, arena.Size.Z) / 2 * SPAWN_RADIUS
	local angle = math.random() * math.pi * 2
	local dist = math.sqrt(math.random()) * radius
	local c = arena.Position
	return Vector3.new(c.X + math.cos(angle) * dist, getArenaTop() + 5, c.Z + math.sin(angle) * dist)
end

local function spawnEnemy()
	local enemy = enemyTemplate:Clone()
	enemy.Name = "Enemy"
	enemy:PivotTo(CFrame.new(getRandomPosition()))
	enemy.Parent = workspace

	-- KLUCZOWE: serwer jest właścicielem fizyki bota,
	-- inaczej ApplyImpulse z ataku gracza często nic nie robi
	local root = enemy:FindFirstChild("HumanoidRootPart")
	if root then
		root:SetNetworkOwner(nil)
	end

	local hum = enemy:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.Died:Connect(function()
			task.wait(1)
			if enemy.Parent then enemy:Destroy() end
		end)
	end
end

-- Główna pętla
while true do
	local count = 0
	local limitY = getArenaTop() - FALL_LIMIT

	for _, obj in ipairs(workspace:GetChildren()) do
		if obj:IsA("Model") and obj.Name == "Enemy" then
			local root = obj:FindFirstChild("HumanoidRootPart")
			if not root or root.Position.Y < limitY then
				obj:Destroy() -- spadł z areny
			else
				count += 1
			end
		end
	end

	for i = 1, MAX_ENEMIES - count do
		spawnEnemy()
		task.wait(0.2)
	end

	task.wait(1)
end

2. AttackHandlerScript

ServerScriptService → AttackHandler

Serwer odbiera kliknięcie gracza i odpycha boty w zasięgu (z ogłuszeniem).

-- MIEJSCE: ServerScriptService > AttackHandler  (typ: Script)
-- Odbiera atak od gracza i odpycha boty w zasięgu.

local Players = game:GetService("Players")
local event = game.ReplicatedStorage:WaitForChild("AttackEvent")

-- KONFIGURACJA
local ATTACK_DIST = 12     -- zasięg ataku (było 18)
local ATTACK_POWER = 100   -- siła uderzenia (było 150)
local STUN_TIME = 0.6      -- ile sekund bot "leci" bezwładnie po trafieniu
local COOLDOWN = 0.5       -- zabezpieczenie po stronie serwera (przed spamem)

local lastAttack = {}

event.OnServerEvent:Connect(function(player)
	local now = os.clock()
	if lastAttack[player] and now - lastAttack[player] < COOLDOWN then return end
	lastAttack[player] = now

	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local myHum = char and char:FindFirstChildOfClass("Humanoid")
	if not root or not myHum or myHum.Health <= 0 then return end

	for _, obj in ipairs(workspace:GetChildren()) do
		if obj:IsA("Model") and obj.Name == "Enemy" then
			local eRoot = obj:FindFirstChild("HumanoidRootPart")
			local eHum = obj:FindFirstChildOfClass("Humanoid")

			if eRoot and eHum and eHum.Health > 0 then
				local offset = eRoot.Position - root.Position
				local flat = Vector3.new(offset.X, 0, offset.Z)

				if flat.Magnitude <= ATTACK_DIST and flat.Magnitude > 0.01 then
					local push = (flat.Unit + Vector3.new(0, 0.35, 0)).Unit

					-- Ogłuszenie: Humanoid przestaje "walczyć" z odrzutem
					local stunEnd = os.clock() + STUN_TIME
					obj:SetAttribute("StunnedUntil", stunEnd)
					eHum.PlatformStand = true

					eRoot:ApplyImpulse(push * eRoot.AssemblyMass * ATTACK_POWER)

					task.delay(STUN_TIME, function()
						-- zdejmujemy ogłuszenie tylko jeśli nie było nowego trafienia
						if obj.Parent and (obj:GetAttribute("StunnedUntil") or 0) <= os.clock() then
							eHum.PlatformStand = false
						end
					end)
				end
			end
		end
	end
end)

Players.PlayerRemoving:Connect(function(p)
	lastAttack[p] = nil
end)

3. Atak graczaLocalScript

StarterPlayer → StarterCharacterScripts → PlayerAttack

Mysz, klawisz F, pad (R2) i przycisk „PCHNIJ” na telefonie.

-- MIEJSCE: StarterPlayer > StarterCharacterScripts > PlayerAttack  (typ: LocalScript)
-- Atak: lewy przycisk myszy / klawisz F / R2 na padzie / przycisk na telefonie.

local ContextActionService = game:GetService("ContextActionService")
local event = game.ReplicatedStorage:WaitForChild("AttackEvent")
local humanoid = script.Parent:WaitForChild("Humanoid")

local COOLDOWN = 0.5
local canAttack = true

local function attack(_, state)
	if state ~= Enum.UserInputState.Begin then
		return Enum.ContextActionResult.Pass
	end
	if canAttack and humanoid.Health > 0 then
		canAttack = false
		event:FireServer() -- informujemy serwer o ataku
		task.delay(COOLDOWN, function() canAttack = true end)
	end
	return Enum.ContextActionResult.Pass
end

ContextActionService:BindAction("SumoAttack", attack, true,
	Enum.UserInputType.MouseButton1, Enum.KeyCode.F, Enum.KeyCode.ButtonR2)
ContextActionService:SetTitle("SumoAttack", "PCHNIJ")

humanoid.Died:Connect(function()
	ContextActionService:UnbindAction("SumoAttack")
end)

4. LawaScript

Workspace → Baseplate → LavaScript

Gracz w lawie ginie, bot znika (Manager robi nowego).

-- MIEJSCE: Workspace > Baseplate > LavaScript  (typ: Script)
-- Gracz w lawie ginie, bot w lawie znika (Manager stworzy nowego).

local lava = script.Parent

lava.Touched:Connect(function(hit)
	local character = hit:FindFirstAncestorOfClass("Model")
	if not character then return end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid or humanoid.Health <= 0 then return end

	if game.Players:GetPlayerFromCharacter(character) then
		humanoid.Health = 0
	elseif character.Name == "Enemy" then
		character:Destroy()
	end
end)

5. AI botaScript

ServerStorage → Enemy → EnemyAI

Bot goni najbliższego gracza i odpycha go przy dotyku.

-- MIEJSCE: ServerStorage > Enemy > EnemyAI  (typ: Script)
-- AI bota: goni najbliższego gracza i odpycha go przy dotyku.

local Players = game:GetService("Players")
local npc = script.Parent
local hum = npc:WaitForChild("Humanoid")
local root = npc:WaitForChild("HumanoidRootPart")

-- KONFIGURACJA
local PUSH_POWER = 50      -- siła odrzutu gracza
local PUSH_DURATION = 0.2  -- jak długo trwa odrzut
local SEARCH_RANGE = 100   -- zasięg wzroku bota
local HIT_COOLDOWN = 0.8   -- bot nie może pchać tego samego gracza częściej

local lastHit = {}

local function isStunned()
	return (npc:GetAttribute("StunnedUntil") or 0) > os.clock()
end

local function findTarget()
	local closest, shortest = nil, SEARCH_RANGE
	for _, player in ipairs(Players:GetPlayers()) do
		local char = player.Character
		local pRoot = char and char:FindFirstChild("HumanoidRootPart")
		local pHum = char and char:FindFirstChildOfClass("Humanoid")
		if pRoot and pHum and pHum.Health > 0 then
			local dist = (pRoot.Position - root.Position).Magnitude
			if dist < shortest then
				shortest = dist
				closest = pRoot
			end
		end
	end
	return closest
end

root.Touched:Connect(function(hit)
	if isStunned() or hum.Health <= 0 then return end

	local char = hit:FindFirstAncestorOfClass("Model")
	local player = char and Players:GetPlayerFromCharacter(char)
	if not player then return end

	local pRoot = char:FindFirstChild("HumanoidRootPart")
	if not pRoot or pRoot:FindFirstChild("PushForce") then return end
	if lastHit[player] and os.clock() - lastHit[player] < HIT_COOLDOWN then return end
	lastHit[player] = os.clock()

	local flat = Vector3.new(pRoot.Position.X - root.Position.X, 0, pRoot.Position.Z - root.Position.Z)
	if flat.Magnitude < 0.01 then flat = root.CFrame.LookVector end
	local pushVector = (flat.Unit + Vector3.new(0, 0.5, 0)).Unit * PUSH_POWER

	local attachment = Instance.new("Attachment")
	attachment.Name = "PushAttachment"
	attachment.Parent = pRoot

	local force = Instance.new("LinearVelocity")
	force.Name = "PushForce"
	force.Attachment0 = attachment
	force.RelativeTo = Enum.ActuatorRelativeTo.World
	force.MaxForce = math.huge
	force.VectorVelocity = pushVector
	force.Parent = pRoot

	root.AssemblyLinearVelocity = Vector3.zero -- bot zatrzymuje się po uderzeniu

	task.delay(PUSH_DURATION, function()
		force:Destroy()
		attachment:Destroy()
	end)
end)

-- Pętla ruchu
while npc.Parent and hum.Health > 0 do
	if not isStunned() then
		local target = findTarget()
		hum:MoveTo(target and target.Position or root.Position)
	end
	task.wait(0.3)
end

6. NPCAnimateScript

ServerStorage → Enemy → NPCAnimate

Animacje chodzenia, stania i spadania bota.

-- MIEJSCE: ServerStorage > Enemy > NPCAnimate  (typ: Script)
-- Animacje bota (LocalScript "Animate" w bocie by nie działał, dlatego zwykły Script).

local npc = script.Parent
local hum = npc:WaitForChild("Humanoid")
-- bot z Rig Buildera może nie mieć Animatora, więc tworzymy go w razie potrzeby
local animator = hum:FindFirstChildOfClass("Animator")
if not animator then
	animator = Instance.new("Animator")
	animator.Parent = hum
end

local function load(id, looped)
	local anim = Instance.new("Animation")
	anim.AnimationId = "rbxassetid://" .. id
	local track = animator:LoadAnimation(anim)
	track.Looped = looped
	return track
end

-- standardowe animacje R15 (te same ID co w Twoim Animate)
local idle = load(507766388, true)
local walk = load(507777826, true)
local fall = load(507767968, true)

idle:Play()

hum.Running:Connect(function(speed)
	if speed > 0.5 then
		if not walk.IsPlaying then walk:Play(0.2) end
		walk:AdjustSpeed(math.clamp(speed / 14, 0.5, 1.5))
	elseif walk.IsPlaying then
		walk:Stop(0.2)
	end
end)

hum.StateChanged:Connect(function(_, new)
	if new == Enum.HumanoidStateType.Freefall or new == Enum.HumanoidStateType.PlatformStanding then
		if not fall.IsPlaying then fall:Play(0.1) end
	elseif fall.IsPlaying then
		fall:Stop(0.2)
	end
end)
